"use client";
import React, { useState } from "react";

export default function CorporateArtForm() {
  const [submitting, setSubmitting] = useState(false);
  const [status, setStatus] = useState<null | { ok: boolean; msg: string }>(null);

  async function handleSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    setSubmitting(true);
    setStatus(null);

    const form = e.currentTarget;
    const data = new FormData(form);

    // Basic client-side validation
    if (!data.get("name") || !data.get("company") || !data.get("email")) {
      setSubmitting(false);
      setStatus({ ok: false, msg: "Please fill in your name, company, and email." });
      return;
    }

    try {
      const res = await fetch("https://formspree.io/f/xqayrpoe", {
        method: "POST",
        headers: { Accept: "application/json" },
        body: data
      });

      if (res.ok) {
        form.reset();
        setStatus({ ok: true, msg: "Thanks! Your inquiry has been sent. I’ll reply soon." });
      } else {
        const err = await res.json().catch(() => ({}));
        setStatus({
          ok: false,
          msg: err?.errors?.[0]?.message || "Something went wrong. Please try again."
        });
      }
    } catch {
      setStatus({ ok: false, msg: "Network error. Please try again." });
    } finally {
      setSubmitting(false);
    }
  }

  return (
    <form
      onSubmit={handleSubmit}
      className="max-w-xl mx-auto p-6 bg-white rounded-2xl shadow-md space-y-4"
    >
      <h2 className="text-2xl font-semibold text-gray-900">Let’s Talk About Your Space</h2>
      <p className="text-gray-600 text-sm">Tell me a bit about your project — I’ll follow up soon.</p>

      <label className="block">
        <span className="text-gray-700">Full Name *</span>
        <input name="name" required className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Company / Organization *</span>
        <input name="company" required className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Email *</span>
        <input name="email" type="email" required className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Phone (optional)</span>
        <input name="phone" type="tel" className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Type of Project</span>
        <select name="projectType" className="mt-1 block w-full border rounded-md p-2">
          <option>Office / Corporate HQ</option>
          <option>Hotel / Resort</option>
          <option>Healthcare / Wellness</option>
          <option>Real Estate Development</option>
          <option>Restaurant / Hospitality</option>
          <option>Other</option>
        </select>
      </label>

      <label className="block">
        <span className="text-gray-700">Estimated Budget Range</span>
        <select name="budget" className="mt-1 block w-full border rounded-md p-2">
          <option>Under $5,000</option>
          <option>$5,000–$15,000</option>
          <option>$15,000–$50,000</option>
          <option>$50,000+</option>
        </select>
      </label>

      <label className="block">
        <span className="text-gray-700">Number of Pieces</span>
        <input name="pieces" type="number" min={0} className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <fieldset className="block">
        <legend className="text-gray-700 mb-2">Art Preference</legend>
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-2 text-sm">
          <label className="flex items-center gap-2"><input type="checkbox" name="style" value="Abstract" /> Abstract</label>
          <label className="flex items-center gap-2"><input type="checkbox" name="style" value="Nature-Inspired" /> Nature-Inspired</label>
          <label className="flex items-center gap-2"><input type="checkbox" name="style" value="Textural / Mixed Media" /> Textural / Mixed Media</label>
          <label className="flex items-center gap-2"><input type="checkbox" name="style" value="Custom Commission" /> Custom Commission</label>
          <label className="flex items-center gap-2"><input type="checkbox" name="style" value="Unsure / Open" /> Unsure / Open to Ideas</label>
        </div>
      </fieldset>

      <label className="block">
        <span className="text-gray-700">Project Location</span>
        <input name="location" className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Deadline / Timeline</span>
        <input name="timeline" placeholder="e.g., Installation by March 2026" className="mt-1 block w-full border rounded-md p-2" />
      </label>

      <label className="block">
        <span className="text-gray-700">Message / Vision</span>
        <textarea name="message" rows={4} className="mt-1 block w-full border rounded-md p-2"></textarea>
      </label>

      {/* Metadata + spam trap */}
      <input type="hidden" name="_subject" value="New Corporate Art Inquiry" />
      <input type="hidden" name="source" value="Corporate Art Buyers Page" />
      <input type="text" name="_gotcha" className="hidden" tabIndex={-1} autoComplete="off" />

      <label className="flex items-start gap-2 text-sm text-gray-600">
        <input type="checkbox" name="consent" value="yes" className="mt-1" />
        I agree to be contacted about this inquiry. I don’t sell or share your info.
      </label>

      <button
        type="submit"
        disabled={submitting}
        className="bg-black text-white px-4 py-2 rounded-md hover:bg-gray-800 disabled:opacity-50"
      >
        {submitting ? "Sending…" : "Send Inquiry"}
      </button>

      {status && (
        <p className={`text-sm ${status.ok ? "text-green-700" : "text-red-700"}`}>
          {status.msg}
        </p>
      )}

      <p className="text-xs text-gray-500">I typically respond within 24–48 hours.</p>
    </form>
  );
}
